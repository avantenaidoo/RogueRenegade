import { useEffect, useRef, useState } from "react";
import { invoke } from "@tauri-apps/api/core";
import { listen } from "@tauri-apps/api/event";
import { Terminal } from "xterm";
import { FitAddon } from "xterm-addon-fit";
import "xterm/css/xterm.css";
import { useEditorStore } from "../../store";
import TerminalSettings from "./TerminalSettings";

export default function TerminalPanel({
  workspaceId,
  workspaceRoot,
}: {
  workspaceId: string | null;
  workspaceRoot: string | null;
}) {
  const {
    terminals,
    activeTerminalId,
    addTerminal,
    setActiveTerminal,
    closeTerminal,
    toggleTerminalPanel,
  } = useEditorStore();

  const terminalIdRef = useRef<string | null>(null);
  const activeIdRef = useRef<string | null>(null);
  const unlistenRef = useRef<(() => void) | null>(null);

  const containerRef = useRef<HTMLDivElement>(null);
  const termRef = useRef<Terminal | null>(null);
  const fitRef = useRef<FitAddon | null>(null);

  const didSetupRef = useRef(false);
  const [isMaximized, setIsMaximized] = useState(false);

  // ─────────────────────────────────────────────
  // SAFE TERMINAL ID CHECK
  // ─────────────────────────────────────────────
  const getValidId = () => {
    const { terminals } = useEditorStore.getState();
    if (!activeIdRef.current) return null;
    return terminals.some(t => t.id === activeIdRef.current)
      ? activeIdRef.current
      : null;
  };

  // Sync active ref
  useEffect(() => {
    const unsubscribe = useEditorStore.subscribe((state) => {
      activeIdRef.current = state.activeTerminalId;
    });
    return unsubscribe;
  }, []);

  // ─────────────────────────────────────────────
  // HARD RESET WHEN ALL TERMINALS CLOSED
  // ─────────────────────────────────────────────
  useEffect(() => {
    if (terminals.length === 0) {
      didSetupRef.current = false;

      if (unlistenRef.current) {
        unlistenRef.current();
        unlistenRef.current = null;
      }

      if (termRef.current) {
        termRef.current.dispose();
        termRef.current = null;
        fitRef.current = null;
      }

      activeIdRef.current = null;
    }
  }, [terminals.length]);

  // ─────────────────────────────────────────────
  // SETUP TERMINAL
  // ─────────────────────────────────────────────
  useEffect(() => {
    if (!workspaceId || !containerRef.current) return;
    if (terminals.length === 0) return;
    if (didSetupRef.current) return;

    didSetupRef.current = true;

    const term = new Terminal({
      cursorBlink: true,
      cursorStyle: "block",
      fontFamily: "monospace",
      fontSize: 14,
      lineHeight: 1.2,
      theme: { background: "#1e1e1e", foreground: "#cccccc" },
    });

    const fit = new FitAddon();
    term.loadAddon(fit);
    term.open(containerRef.current);
    requestAnimationFrame(() => fit.fit());

    termRef.current = term;
    fitRef.current = fit;

    term.onData((data) => {
      const id = getValidId();
      if (!id) return;
      invoke("terminal_write", { id, data });
    });

    (async () => {
      const existing = terminals.find((t) => t.workspaceId === workspaceId);

      let id: string;
      let shell: string;

      if (existing) {
        id = existing.id;
        shell = existing.shell;
      } else {
        [id, shell] = await invoke<[string, string]>("terminal_spawn", {
          shell: null,
          cwd: workspaceRoot ?? undefined,
        });
        addTerminal({ id, shell, workspaceId: workspaceId!, active: true });
      }

      terminalIdRef.current = id;
      activeIdRef.current = id;
      setActiveTerminal(id);

      if (unlistenRef.current) unlistenRef.current();
      unlistenRef.current = await listen<[string, string]>(
        "terminal-output",
        (e) => {
          if (e.payload[0] !== activeIdRef.current) return;
          termRef.current?.write(e.payload[1]);
        }
      );

      await invoke("terminal_ready", { id });

      requestAnimationFrame(() => {
        fit.fit();
        invoke("terminal_resize", {
          id,
          rows: term.rows,
          cols: term.cols,
        });
      });
    })();

    return () => {
      if (unlistenRef.current) unlistenRef.current();
    };
  }, [workspaceId, workspaceRoot, terminals]);

  // ─────────────────────────────────────────────
  // TAB SWITCH FIX
  // ─────────────────────────────────────────────
  useEffect(() => {
    if (!activeTerminalId || !termRef.current || !fitRef.current) return;

    activeIdRef.current = activeTerminalId;

    termRef.current.reset(); // clears old buffer

    requestAnimationFrame(() => {
      fitRef.current?.fit();
      invoke("terminal_resize", {
        id: activeTerminalId,
        rows: termRef.current!.rows,
        cols: termRef.current!.cols,
      });
      invoke("terminal_redraw", { id: activeTerminalId });
    });
  }, [activeTerminalId]);

  // ─────────────────────────────────────────────
  // RESIZE OBSERVER SAFE
  // ─────────────────────────────────────────────
  useEffect(() => {
    if (!containerRef.current) return;

    const resize = () => {
      requestAnimationFrame(() => {
        if (!fitRef.current || !termRef.current) return;
        const id = getValidId();
        if (!id) return;

        fitRef.current.fit();
        invoke("terminal_resize", {
          id,
          rows: termRef.current.rows,
          cols: termRef.current.cols,
        });
      });
    };

    resize();
    const observer = new ResizeObserver(resize);
    observer.observe(containerRef.current);
    return () => observer.disconnect();
  }, []);

  const spawnTerminal = async (shell: string | null) => {
    const [id, resolvedShell] = await invoke<[string, string]>(
      "terminal_spawn",
      { shell, cwd: workspaceRoot ?? undefined }
    );

    const fallbackWsId = workspaceId ?? `blank-${Date.now()}`;
    useEditorStore.getState().addTerminal({
      id,
      shell: resolvedShell,
      workspaceId: fallbackWsId,
      active: true,
    });

    useEditorStore.getState().setActiveTerminal(id);
  };

  // ─────────────────────────────────────────────
  // YOUR ORIGINAL UI BELOW — UNCHANGED
  // ─────────────────────────────────────────────

  if (terminals.length === 0) {
    return (
      <div className="flex flex-col w-full h-full bg-default border-t border-default">
        <div className="flex items-center justify-between px-2 h-9 bg-muted border-b border-default">
          <div className="flex gap-3 text-xs text-muted">
            {["PROBLEMS", "OUTPUT", "DEBUG CONSOLE", "PORTS", "TERMINAL"].map(v => (
              <button key={v} className={v === "TERMINAL" ? "text-default font-medium" : ""}>
                {v}
              </button>
            ))}
          </div>
          <div className="flex items-center gap-2 text-muted">
            <TerminalSettings onSpawn={spawnTerminal} />
            <button onClick={toggleTerminalPanel}>✕</button>
          </div>
        </div>
        <div className="flex-1 bg-black flex items-center justify-center text-muted">
          No terminals open. Click + to create one.
        </div>
      </div>
    );
  }

  return (
    <div className={`flex flex-col w-full h-full bg-default border-t border-default ${
      isMaximized ? 'fixed top-0 left-0 w-full h-full z-50' : ''
    }`}>
      <div className="flex items-center justify-between px-2 h-9 bg-muted border-b border-default">
        <div className="flex gap-3 text-xs text-muted">
          {["PROBLEMS", "OUTPUT", "DEBUG CONSOLE", "PORTS", "TERMINAL"].map(v => (
            <button key={v} className={v === "TERMINAL" ? "text-default font-medium" : ""}>
              {v}
            </button>
          ))}
        </div>

        <div className="flex items-center gap-2 text-muted">
          <TerminalSettings onSpawn={spawnTerminal} />
          <button onClick={() => setIsMaximized(prev => !prev)}>
            {isMaximized ? '🗗' : '▢'}
          </button>
          <button onClick={toggleTerminalPanel}>✕</button>
        </div>
      </div>

      <div className="flex flex-1 overflow-hidden">
        <div ref={containerRef} className="flex-1 bg-black" />

        <div className="w-40 border-l border-default bg-muted text-xs overflow-y-auto">
          {terminals.map(t => (
            <div
              key={t.id}
              className={`
                flex justify-between items-center px-3 h-10 cursor-pointer relative group
                transition-colors duration-150
                ${t.id === activeTerminalId
                  ? "bg-default text-foreground border-l-4 border-l-foreground font-medium"
                  : "text-default hover:bg-neutral-500"
                }
              `}
              onClick={() => {
                setActiveTerminal(t.id);
                activeIdRef.current = t.id;
              }}
            >
              <span className="flex-1 truncate">
                {t.shell}
              </span>

              <button
                className="ml-2 text-red-500 opacity-0 group-hover:opacity-100 transition-opacity duration-150 text-lg"
                onClick={(e) => {
                  e.stopPropagation();
                  invoke("terminal_kill", { id: t.id }).catch(() => {});
                  closeTerminal(t.id);
                }}
              >
                🗑️
              </button>
            </div>
          ))}
        </div>
      </div>
    </div>
  );
}
